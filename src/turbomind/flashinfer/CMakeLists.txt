# FlashInfer-backed kernels for turbomind.
#
# This directory hosts every turbomind operator that delegates to the
# FlashInfer kernel library (loaded as a sibling .so via tvm-ffi at runtime).
# Each subdirectory owns one FlashInfer-backed runtime bridge and follows the
# FMHA pattern: consume a prepared flashinfer-python/JIT-produced .so path at
# configure time, bake that path into a tiny config header, and keep a runtime
# env-var override for deployment.
#
# The FMHA bridge is built by default.  The MoE and dense GEMM bridges target
# the SM10x Blackwell datacenter family (B200/B300) and self-gate on CUDA >= 12.8.
#
# Docker builds prepare FlashInfer JIT/AOT artifacts up front with
# `docker/prepare_flashinfer_artifacts.sh`.  That script writes a tiny CMake
# cache fragment so the subdirectories only need to consume already-resolved
# .so paths instead of compiling FlashInfer during CMake configure.

set(LMDEPLOY_FLASHINFER_ARTIFACTS_CMAKE "$ENV{LMDEPLOY_FLASHINFER_ARTIFACTS_CMAKE}"
    CACHE FILEPATH "Optional CMake fragment generated by prepare_flashinfer_artifacts.sh")
if(NOT LMDEPLOY_FLASHINFER_ARTIFACTS_CMAKE AND DEFINED ENV{HOME})
    set(_lmdeploy_flashinfer_default_artifacts
        "$ENV{HOME}/.cache/flashinfer/lmdeploy_artifacts.cmake")
    if(EXISTS "${_lmdeploy_flashinfer_default_artifacts}")
        set(LMDEPLOY_FLASHINFER_ARTIFACTS_CMAKE
            "${_lmdeploy_flashinfer_default_artifacts}"
            CACHE FILEPATH "Optional CMake fragment generated by prepare_flashinfer_artifacts.sh" FORCE)
    endif()
endif()
if(LMDEPLOY_FLASHINFER_ARTIFACTS_CMAKE AND EXISTS "${LMDEPLOY_FLASHINFER_ARTIFACTS_CMAKE}")
    include("${LMDEPLOY_FLASHINFER_ARTIFACTS_CMAKE}")
    message(STATUS "[flashinfer] loaded artifacts: ${LMDEPLOY_FLASHINFER_ARTIFACTS_CMAKE}")
endif()

function(lmdeploy_flashinfer_env_to_cache var type doc)
    if(NOT DEFINED ${var} OR "${${var}}" STREQUAL "")
        set(_value "$ENV{${var}}")
        if(NOT "${_value}" STREQUAL "")
            set(${var} "${_value}" CACHE ${type} "${doc}" FORCE)
        endif()
    endif()
endfunction()

    "Absolute path to FlashInfer trtllm-gen FMHA .so")

    "Absolute path to FlashInfer trtllm-gen MoE .so")

    "Absolute path to FlashInfer CUTLASS fused-MoE .so")
lmdeploy_flashinfer_env_to_cache(LMDEPLOY_FLASHINFER_GEMM_SO_PATH FILEPATH
    "Absolute path to FlashInfer SM10x GEMM .so")
lmdeploy_flashinfer_env_to_cache(LMDEPLOY_FLASHINFER_FP4_GEMM_SO_PATH FILEPATH
    "Absolute path to FlashInfer SM10x FP4 GEMM .so")
lmdeploy_flashinfer_env_to_cache(LMDEPLOY_FLASHINFER_FP4_GEMM_SM100_SO_PATH FILEPATH
    "Absolute path to FlashInfer SM100 FP4 GEMM .so")
lmdeploy_flashinfer_env_to_cache(LMDEPLOY_FLASHINFER_FP4_GEMM_SM103_SO_PATH FILEPATH
    "Absolute path to FlashInfer SM103 FP4 GEMM .so")
lmdeploy_flashinfer_env_to_cache(LMDEPLOY_FLASHINFER_FMHA_CUBIN_CACHE PATH
    "FlashInfer trtllm-gen FMHA cubin cache root")
lmdeploy_flashinfer_env_to_cache(LMDEPLOY_FLASHINFER_MOE_CUBIN_CACHE PATH
    "FlashInfer trtllm-gen MoE/BMM cubin cache root")
